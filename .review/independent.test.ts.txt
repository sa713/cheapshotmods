const cfg:Config={baseUrl:'http://reviewer.local',sessionSecret:'reviewer-isolated-session-secret-32-characters',botToken:'reviewer-synthetic-bot-token',clientId:'reviewer-client',clientSecret:'reviewer-synthetic-client-secret',version:process.env.REVIEW_TARGET_SHA??'b'.repeat(40)};
function miniData(id='713001') {
  const p=new URLSearchParams({auth_date:String(Math.floor(Date.now()/1000)),user:JSON.stringify({id,first_name:'Independent fixture'}),query_id:randomUUID()});
  const text=[...p.entries()].sort(([a],[b])=>a<b?-1:a>b?1:0).map(([k,v])=>k+'='+v).join('\n');
  const secret=createHmac('sha256','WebAppData').update(cfg.botToken!).digest();
  p.set('hash',createHmac('sha256',secret).update(text).digest('hex'));return p.toString();
}
async function jsonRequest(address:string,path:string,headers:Record<string,string>={},body?:unknown) {
  const r=await fetch(address+path,{headers,method:body===undefined?'GET':'POST',...(body===undefined?{}:{body:JSON.stringify(body)})});
  const text=await r.text();return {status:r.status,text,data:JSON.parse(text),headers:r.headers};
}
async function login(address:string,id='713001') {
  const r=await jsonRequest(address,'/auth/mini',{'content-type':'application/json',origin:cfg.baseUrl},{initData:miniData(id)});
  assert.equal(r.status,200,r.text);return {cookie:r.headers.get('set-cookie')!.split(';')[0]!,origin:cfg.baseUrl,'x-csrf-token':r.data.csrf,'content-type':'application/json'};
}
async function command(address:string,headers:Record<string,string>,action:unknown) {
  const s=await jsonRequest(address,'/api/state',headers);assert.equal(s.status,200);
  const r=await jsonRequest(address,'/api/commands',headers,{key:randomUUID(),revision:s.data.revision,experiment:s.data.experiment,action});
  assert.equal(r.status,200,r.text);return r.data;
}
function snapshot(s:Store) {
  return ['players','groups','history','commands'].map(t=>s.db.prepare('SELECT * FROM '+t+' ORDER BY rowid').all());
}

test('independent OIDC cryptographic and time boundary matrix',async()=>{
  const now=1900000000,pair=await generateKeyPair('RS256'),jwks=createLocalJWKSet({keys:[{...await exportJWK(pair.publicKey),alg:'RS256',kid:'reviewer'}]});
  const remote:Transport={jwks,exchange:async()=>{throw new Error('not used');}};
  const signed=(extra:Record<string,unknown>={})=>new SignJWT({iss:issuer,aud:cfg.clientId,sub:'distinct-sub',id:'713001',iat:now,exp:now+600,nonce:'n',...extra}).setProtectedHeader({alg:'RS256',kid:'reviewer'}).sign(pair.privateKey);
  for(const delta of [-3600,0,5,30]) assert.equal(await oidcIdentity(await signed({iat:now+delta}),'n',cfg,remote,new Date(now*1000)),'713001');
  for(const extra of [{iat:now+31},{iat:now-3601},{exp:now},{exp:now-1},{iat:now+5,exp:now+5},{nbf:now+1},{nonce:'other'},{iss:'https://evil.invalid'},{aud:'other'},{id:undefined},{nonce:undefined}]) {
    await assert.rejects(()=>signed(extra).then(t=>oidcIdentity(t,'n',cfg,remote,new Date(now*1000))),JSON.stringify(extra));
  }
  assert.equal(await oidcIdentity(await signed({nbf:now}),'n',cfg,remote,new Date(now*1000)),'713001');
  const badPair=await generateKeyPair('RS256');
  const bad=await new SignJWT({iss:issuer,aud:cfg.clientId,sub:'s',id:'713001',iat:now,exp:now+600,nonce:'n'}).setProtectedHeader({alg:'RS256',kid:'reviewer'}).sign(badPair.privateKey);
  await assert.rejects(()=>oidcIdentity(bad,'n',cfg,remote,new Date(now*1000)));
});

      const before=snapshot(s);const r=await fetch(address+'/api/commands',{method:'POST',headers:{...h,'content-type':content},body});

test('independent real HTTP 128 digit boundary, million digit rejection, exact 129 digit craft lifecycle',async()=>{
  const s=new Store(':memory:'),f=await app(s,cfg),address=await f.listen({host:'127.0.0.1',port:0});
  try {
    const h=await login(address),other=await login(address,'713002'),huge='9'.repeat(128),tooLong='9'.repeat(129);
    const initial=await command(address,h,{type:'original.put',kind:'💚',tier:1,value:huge,quantity:'9'});
    const hugeQuantity=await command(address,h,{type:'original.put',kind:'🧿',tier:6,value:'9007199254740993',quantity:huge});
    const normal=await command(address,h,{type:'level',value:'9007199254740993'});
    const state=(await jsonRequest(address,'/api/state',h)).data;
    const base={key:randomUUID(),revision:state.revision,experiment:state.experiment,action:{type:'level',value:'200'}};
    const inputs=[
      {...base,revision:tooLong},
      {...base,action:{type:'level',value:tooLong}},
      {...base,action:{type:'level',value:'9'.repeat(1000000)}},
      {...base,action:{type:'original.put',kind:'💚',tier:1,value:tooLong,quantity:'1'}},
      {...base,action:{type:'original.put',kind:'💚',tier:1,value:'1',quantity:tooLong}},

      const r=await jsonRequest(address,'/api/commands',h,input);assert.equal(r.status,400,r.text);assert.equal(r.data.error,'INTEGER_TOO_LONG');assert.deepEqual(snapshot(s),before);

    await command(address,h,{type:'level',value:normal.level});
    const crafted=[];
    for(let i=0;i<3;i++) crafted.push(await command(address,h,{type:'craft',items:[{id:initial.createdId,quantity:'3',worn:'0'}]}));
    const v=((BigInt(huge)*3n+1n)/2n).toString();assert.equal(v.length,129);
    const work=(await jsonRequest(address,'/api/inventory?scope=work',h)).data.items;
    for(const c of crafted) {assert.equal(work.find((g:{id:string})=>g.id===c.createdId).value,v);}
    assert.equal(work.find((g:{id:string})=>g.id===hugeQuantity.createdId).quantity,huge);
    const equipped=await command(address,h,{type:'equip',items:[{id:crafted[0].createdId,quantity:'1'}]});
    assert.equal(equipped.stats.health,((BigInt(normal.level)*(100n+BigInt(v))+99n)/100n).toString());
    const next=await command(address,h,{type:'craft',items:crafted.map((c,i)=>({id:c.createdId,quantity:'1',worn:i===0?'1':'0'}))});
    const nextV=((BigInt(v)*3n+1n)/2n).toString();assert.deepEqual(next.equipment,[]);assert.equal(next.lastCraft.output.value,nextV);assert.equal(next.spent,'324');
    await command(address,h,{type:'equip',items:[{id:next.createdId,quantity:'1'}]});
    const undone=await command(address,h,{type:'undo'});assert.equal(undone.spent,'162');assert.deepEqual(undone.equipment,[{id:crafted[0].createdId,quantity:'1'}]);
